### 数据库表单设计
```sql
CREATE TABLE products (
id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,  -- 自增主键
sku VARCHAR(20) NOT NULL UNIQUE,             -- 业务唯一标识
name VARCHAR(100) NOT NULL,
barcode VARCHAR(20) NOT NULL UNIQUE,
category_id INT NOT NULL,
supplier_id INT NOT NULL,
unit_price DECIMAL(10,2) UNSIGNED,
safe_stock INT UNSIGNED DEFAULT 0,
expiry_days SMALLINT UNSIGNED,              -- 保质期天数
is_perishable BOOL DEFAULT false,           -- 是否需保质期管理
status ENUM('NORMAL','DISCONTINUED') DEFAULT 'NORMAL',
created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 扩展属性表（EAV模式）
CREATE TABLE product_attributes (
product_id INT UNSIGNED,
attr_key VARCHAR(50),
attr_value TEXT,
PRIMARY KEY(product_id, attr_key)
);
```
### Cpp 对象模型
```cpp
class Product {
private:
    std::string sku;
    std::string name;
    int categoryId;
    double unitPrice;
    // 使用SQLPP的日期处理
    sqlpp::chrono::time_point productionDate; 
public:
    // 条码校验静态方法
    static bool validateBarcode(const std::string& code) {
        return std::regex_match(code, std::regex("^\\d{12,13}$"));
    }
};
```
